-- ==========================================
-- 火球 (Fireball)
-- ID: 1004
-- ==========================================

local c1004 = {}

function c1004.initial_effect(c)
    local e1 = Effect.CreateEffect(c)
    e1:SetType(EffectType.IGNITION)
    e1:SetRange(Location.HAND)
    e1:SetOperation(c1004.activate)
    c:RegisterEffect(e1)
end

function c1004.activate(e, tp, eg, ep, ev, re, r, rp)
    Duel.SelectTarget(tp, Location.MZONE, 0, 1)
    local target = Duel.GetFirstTarget()
    if target then
        Duel.Log("【火球】发动！造成 5 点伤害。")
        Duel.Damage(target, 5)
    end
end

return c1004
