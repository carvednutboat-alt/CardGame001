-- c1001.lua: Damage Spell
local cm = {}

function cm.initial_effect(c)
    -- Register Activate Effect
    local e1 = Effect()
    e1:SetDescription("Deal 5 damage to a target.")
    -- Type: Activate on Click (Ignition-like)
    -- Code: EFFECT_TYPE_ACTIVATE (We haven't defined codes yet, but conceptually)
    
    -- Condition: Can be activated? (Chk=0)
    e1:SetCondition(cm.condition)
    
    -- Cost: None (or Mana)
    -- e1:SetCost(cm.cost)

    -- Target: Select 1 Enemy
    e1:SetTarget(cm.target)

    -- Operation: Deal Damage
    e1:SetOperation(cm.operation)

    c:register_effect(e1)
end

function cm.condition(e, tp, eg, ep, ev, re, r, rp)
    -- Check if there are valid targets (simplified)
    -- In full YGO, we check Duel.IsExistingTarget(...)
    return true
end

function cm.target(e, tp, eg, ep, ev, re, r, rp, chk)
    -- Calling Duel.SelectTarget initiates the Targeting Mode in C#
    -- Arguments: Card, TargetType
    -- 1 = Enemy, 2 = Ally, 3 = All (Simple Enum mapping)
    Duel.SelectTarget(e:GetHandler(), 1) 
end

function cm.operation(e, tp, eg, ep, ev, re, r, rp)
    -- Get the selected target
    local tc = Duel.GetFirstTarget()
    if tc then
        Duel.Log("Lua: Dealing damage to " .. tc.Name)
        Duel.Damage(tc, 5)
    end
end

return cm
