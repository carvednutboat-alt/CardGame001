-- ==========================================
-- Lua火球 (Lua Fireball)
-- ID: 1001
-- ==========================================

local c1001 = {}

function c1001.initial_effect(c)
    -- 发动效果
    local e1 = Effect.CreateEffect(c)
    e1:SetType(EffectType.IGNITION)
    e1:SetRange(Location.HAND)
    e1:SetOperation(c1001.activate)
    c:RegisterEffect(e1)
end

function c1001.activate(e, tp, eg, ep, ev, re, r, rp)
    -- 选择一个敌人
    Duel.SelectTarget(tp, Location.MZONE, 0, 1)
    local target = Duel.GetFirstTarget()
    if target then
        Duel.Log("【Lua火球】发动！对 " .. target.Name .. " 造成 5 点伤害。")
        Duel.Damage(target, 5)
    end
end

return c1001
