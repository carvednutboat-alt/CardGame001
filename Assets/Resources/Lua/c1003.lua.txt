-- ==========================================
-- 死者苏生 (Death Reborn)
-- ID: 1003
-- ==========================================

local c1003 = {}

function c1003.initial_effect(c)
    local e1 = Effect.CreateEffect(c)
    e1:SetType(EffectType.IGNITION)
    e1:SetRange(Location.HAND)
    e1:SetOperation(c1003.activate)
    c:RegisterEffect(e1)
end

function c1003.activate(e, tp, eg, ep, ev, re, r, rp)
    -- 为一个友方单位恢复 10 生命
    Duel.SelectTarget(tp, Location.MZONE, 1, 0)
    local target = Duel.GetFirstTarget()
    if target then
        Duel.Log("【死者苏生】发动！为 " .. target.Name .. " 恢复 10 点生命。")
        CS.CombatManager.Instance:Heal(target, 10)
    end
end

return c1003
