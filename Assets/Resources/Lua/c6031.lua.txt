local c6031 = {}

function c6031.initial_effect(c)
    local e1 = Effect.CreateEffect(c)
    e1:SetType(EffectType.IGNITION)
    e1:SetRange(Location.HAND)
    e1:SetOperation(c6031.activate)
    c:RegisterEffect(e1)
end

function c6031.activate(e, tp, eg, ep, ev, re, r, rp)
    local card = e:GetHandler()
    local bm = CS.BattleManager.Instance
    
    if card.IsPendingResolved == true then
        -- EXECUTE
        bm.UIManager:Log("Arcane Nova BLAST!")
        -- Deal 12 to 3 random targets (AOE All Player Units logic from old script)
        local units = bm.UnitManager.PlayerUnits
        for i=0, units.Count-1 do
            bm.CombatManager:ApplyDamage(units[i], 12)
        end
        bm.PlayerUnit:TakeDamage(12)
        
        card.IsPendingResolved = false 
    else
        -- QUEUE
        bm.EnemyManager:AddPendingSpell(card, 2, bm.EnemyManager:FindOwner(card)) 
    end
end

return c6031
