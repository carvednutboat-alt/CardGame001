-- ==========================================
-- 强欲之壶 (Pot of Greed)
-- ID: 1002
-- ==========================================

local c1002 = {}

function c1002.initial_effect(c)
    -- 激活效果: 抽2张牌
    local e1 = Effect.CreateEffect(c)
    e1:SetType(EffectType.IGNITION)
    e1:SetOperation(c1002.activate)
    c:RegisterEffect(e1)
end

function c1002.activate(e, tp, eg, ep, ev, re, r, rp)
    Duel.Log("【强欲之壶】发动！抽 2 张牌。")
    BattleManager.Instance.DeckManager:DrawCards(2)
end

return c1002
