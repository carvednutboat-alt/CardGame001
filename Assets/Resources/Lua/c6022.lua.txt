local c6022 = {}

function c6022.initial_effect(c)
    -- Activate: All Enemies +1/+1
    local e1 = Effect.CreateEffect(c)
    e1:SetType(EffectType.IGNITION)
    e1:SetRange(Location.HAND)
    e1:SetOperation(c6022.activate)
    c:RegisterEffect(e1)
end

function c6022.activate(e, tp, eg, ep, ev, re, r, rp)
    Duel.Log("Pack Tactics: All Enemies +1/+1")
    local bm = CS.BattleManager.Instance
    
    -- "Enemies" context: if Player uses it, it buffs Enemies? 
    -- Assuming this is an Enemy Spell. 
    -- Logic: Buff "My Side" (which for Enemy is EnemyManager.ActiveEnemies)
    -- Safe fallback: Buff all units in EnemyManager.ActiveEnemies
    
    local enemies = bm.EnemyManager.ActiveEnemies
    for i=0, enemies.Count-1 do
        local enemy = enemies[i]
        enemy.UnitData.CurrentAtk = enemy.UnitData.CurrentAtk + 1
        enemy.UnitData.CurrentHp = enemy.UnitData.CurrentHp + 1
        enemy.UnitData.MaxHp = enemy.UnitData.MaxHp + 1
        
        -- Update UI
        if enemy.UI ~= nil then
            enemy.UI:UpdateAttack()
            enemy.UI:UpdateHP()
        end
    end
end

return c6022
