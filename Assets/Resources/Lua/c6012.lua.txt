local c6012 = {}

function c6012.initial_effect(c)
    -- Activate: Assault (Count Colors -> Damage)
    local e1 = Effect.CreateEffect(c)
    e1:SetType(EffectType.IGNITION)
    e1:SetRange(Location.HAND)
    e1:SetOperation(c6012.activate)
    c:RegisterEffect(e1)
end

function c6012.activate(e, tp, eg, ep, ev, re, r, rp)
    local c = e:GetHandler()
    local bm = CS.BattleManager.Instance
    -- Assuming Enemy usage (owner logic in old script). 
    -- If Player uses it, we check Player Colors. If Enemy, Enemy Colors.
    -- But since this is likely an Enemy Card, we default to EnemyManager checks if owner is enemy.
    
    -- Robust check:
    local owner = c.Owner -- 0=Player, 1=Enemy
    local colors = nil
    
    if owner == 0 then
        colors = bm.UnitManager:GetAliveCommanderColors()
    else
        colors = bm.EnemyManager:GetAliveCommanderColors()
    end

    local count = 0
    if colors:Contains(CS.CardColor.Red) then count = count + 1 end
    if colors:Contains(CS.CardColor.Green) then count = count + 1 end
    if colors:Contains(CS.CardColor.Blue) then count = count + 1 end

    bm.UIManager:Log("Assault! Attacks: " .. count)

    for i=1, count do
        -- Find Targets (Opponent)
        local targets = bm:GetOpponentUnits(owner) 
        
        if targets.Count > 0 then
            local targetUnit = targets[CS.UnityEngine.Random.Range(0, targets.Count)]
            bm.CombatManager:ApplyDamage(targetUnit, 5)
            bm.UIManager:Log("Assault hits " .. targetUnit.Name)
        else
            -- If no units, hit Face
            if owner == 1 then -- Enemy attacking Player
                bm.PlayerUnit:TakeDamage(5)
                bm.UIManager:Log("Assault hits Player!")
            elseif owner == 0 then -- Player attacking Enemy (which face? Enemy doesn't have face usually, just units)
                -- Maybe hit random enemy? Or no-op?
                -- Leaving as no-op or hit Boss if exists.
            end
        end
    end
end

return c6012
