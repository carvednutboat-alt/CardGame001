local c6021 = {}

function c6021.initial_effect(c)
    -- Activate: Summon 2 Wolves
    local e1 = Effect.CreateEffect(c)
    e1:SetType(EffectType.IGNITION)
    e1:SetRange(Location.HAND)
    e1:SetOperation(c6021.activate)
    c:RegisterEffect(e1)
end

function c6021.activate(e, tp, eg, ep, ev, re, r, rp)
    Duel.Log("Summoning 2 Wolves...")
    local bm = CS.BattleManager.Instance
    local wolfId = 6050 
    local data = CS.DevCardLoader.GetCardDataShim(wolfId)
    
    if data ~= nil then
        bm.EnemyManager:CreateEnemyAt(data)
        bm.EnemyManager:CreateEnemyAt(data)
    else
        Duel.Log("Wolf Data (6050) Missing!")
    end
end

return c6021
